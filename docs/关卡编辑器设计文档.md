# 关卡编辑器设计文档

## 需求概述

1. 使用 JSON 数据结构描述关卡信息，包括地图、目标分数、时间、订单池
2. 提供可视化界面来搭建关卡，生成和导出 JSON 数据
3. 支持导入和分享关卡 JSON 文件

## 关卡数据结构

### 设计原则

- **单独描述每个对象**：地板、工作站、玩家都使用独立的配置对象
- **支持扩展属性**：每种类型的对象都可以有自己特有的属性
- **避免字符串地图**：不再使用字符矩阵来表达地图布局
- **支持复杂地板**：如传送带需要方向、速度等属性

### 关卡编辑器页面

关卡编辑器页面和正常游戏的页面不属于同一个路由

- `/` 主页面路由显示正常游戏
- `/editor` 路由显示关卡编辑页面

### 完整数据结构

```typescript
interface LevelConfig {
  // 基础信息
  id: string;                    // 关卡唯一ID
  name: string;                  // 关卡名称
  description?: string;          // 关卡描述
  version: string;               // 数据结构版本号（用于未来兼容）
  
  // 游戏设置
  gameType: 'local-coop' | 'local-versus' | 'online-coop' | 'online-versus';
  duration: number;              // 游戏时长（秒）
  
  // 目标分数（星级）
  scoreTarget: {
    star1: number;               // 1星分数
    star2: number;               // 2星分数
    star3: number;               // 3星分数
    star4?: number;              // 4星分数（可选）
  };
  
  // 地图配置
  map: {
    width: number;               // 地图宽度（格子数）
    height: number;              // 地图高度（格子数）
    floors: FloorConfig[];       // 地板配置列表
  };
  
  // 玩家配置
  players: PlayerSpawn[];
  
  // 工作站配置
  stations: StationConfig[];
  
  // 订单池配置
  orderPool: {
    recipes: string[];           // 可用菜谱ID列表
    maxActiveOrders: number;     // 最大同时订单数
    spawnInterval: number;       // 订单生成间隔（秒）
  };
}

// ==================== 玩家配置 ====================

interface PlayerSpawn {
  id: number;                    // 玩家编号（1或2）
  x: number;                     // X坐标（格子）
  y: number;                     // Y坐标（格子）
  color?: number;                // 玩家颜色（可选，16进制）
}

// ==================== 地板配置 ====================

// 地板配置（基类）
interface BaseFloorConfig {
  type: FloorType;               // 地板类型
  x: number;                     // X坐标（格子）
  y: number;                     // Y坐标（格子）
}

// 地板类型
type FloorType = 
  | 'normal'                     // 普通地板
  | 'wall'                       // 墙壁（不可通行）
  | 'conveyor';                  // 传送带

// 联合类型：所有地板配置
type FloorConfig = NormalFloor | WallFloor | ConveyorFloor;

// 普通地板
interface NormalFloor extends BaseFloorConfig {
  type: 'normal';
  texture?: string;              // 自定义纹理key（可选）
}

// 墙壁
interface WallFloor extends BaseFloorConfig {
  type: 'wall';
  texture?: string;              // 自定义纹理key（可选）
}

// 传送带地板
interface ConveyorFloor extends BaseFloorConfig {
  type: 'conveyor';
  direction: 'up' | 'down' | 'left' | 'right';  // 传送方向
  speed: number;                 // 传送速度（像素/秒）
  texture?: string;              // 自定义纹理key（可选）
}

// ==================== 工作站配置 ====================

// 工作站配置（基类）
interface BaseStationConfig {
  type: StationType;             // 工作站类型
  x: number;                     // X坐标（格子）
  y: number;                     // Y坐标（格子）
  rotation?: number;             // 旋转角度（度数，可选）
}

// 工作站类型枚举
type StationType = 
  | 'counter'                    // 空柜台
  | 'plate-counter'              // 带盘子的柜台
  | 'cut'                        // 切菜板
  | 'pot'                        // 炒锅/煮锅
  | 'sink'                       // 洗碗池
  | 'delivery'                   // 上菜口
  | 'dirty-plate'                // 脏盘子生成点
  | 'trash'                      // 垃圾桶
  | 'fire-extinguisher'          // 灭火器
  | 'mixer'                      // 搅拌器
  | 'ingredient';                // 食材箱

// 联合类型：所有工作站配置
type StationConfig = 
  | CounterStation
  | PlateCounterStation
  | CutStation
  | PotStation
  | SinkStation
  | DeliveryStation
  | DirtyPlateStation
  | TrashStation
  | FireExtinguisherStation
  | MixerStation
  | IngredientStation;

// 空柜台
interface CounterStation extends BaseStationConfig {
  type: 'counter';
}

// 带盘子的柜台
interface PlateCounterStation extends BaseStationConfig {
  type: 'plate-counter';
  infinite?: boolean;            // 是否无限供应盘子（默认true）
  maxPlates?: number;            // 最大盘子数（如果infinite为false）
}

// 切菜板
interface CutStation extends BaseStationConfig {
  type: 'cut';
  cutSpeed?: number;             // 切割速度倍率（默认1.0）
}

// 炒锅/煮锅
interface PotStation extends BaseStationConfig {
  type: 'pot';
  cookSpeed?: number;            // 烹饪速度倍率（默认1.0）
  canBurn?: boolean;             // 是否会烧焦（默认true）
  canFire?: boolean;             // 是否会着火（默认true）
}

// 洗碗池
interface SinkStation extends BaseStationConfig {
  type: 'sink';
  washSpeed?: number;            // 清洗速度倍率（默认1.0）
}

// 上菜口
interface DeliveryStation extends BaseStationConfig {
  type: 'delivery';
  deliveryTime?: number;         // 上菜动画时间（毫秒，默认500）
}

// 脏盘子生成点
interface DirtyPlateStation extends BaseStationConfig {
  type: 'dirty-plate';
  spawnInterval?: number;        // 生成间隔（秒，默认10）
  maxPlates?: number;            // 最大脏盘子数（默认3）
}

// 垃圾桶
interface TrashStation extends BaseStationConfig {
  type: 'trash';
}

// 灭火器
interface FireExtinguisherStation extends BaseStationConfig {
  type: 'fire-extinguisher';
  infinite?: boolean;            // 是否无限使用（默认true）
  capacity?: number;             // 容量（如果infinite为false）
}

// 搅拌器
interface MixerStation extends BaseStationConfig {
  type: 'mixer';
  mixSpeed?: number;             // 搅拌速度倍率（默认1.0）
}

// 食材箱
interface IngredientStation extends BaseStationConfig {
  type: 'ingredient';
  ingredientType: IngredientType; // 食材类型
  infinite?: boolean;             // 是否无限供应（默认true）
  maxCount?: number;              // 最大数量（如果infinite为false）
}

// 食材类型
type IngredientType = 
  | 'tomato'                     // 番茄
  | 'lettuce'                    // 生菜
  | 'rice'                       // 米
  | 'fish'                       // 鱼
  | 'seaweed'                    // 紫菜
  | 'onion'                      // 洋葱
  | 'potato'                     // 土豆
  | 'carrot'                     // 胡萝卜
  | 'egg'                        // 鸡蛋
  | 'flour'                      // 面粉
  | 'meat'                       // 肉
  | 'cheese'                     // 芝士
  | 'chocolate'                  // 巧克力
  | 'burger-bun';                // 汉堡面包
```

## 编辑器界面设计

### 技术架构

编辑器采用 **DOM + Canvas 混合架构**：

- **Canvas 主舞台**：地图编辑区域由 Phaser 全屏渲染，画布尺寸始终跟随浏览器视口
- **DOM 浮窗层**：基础设置、目标分数、工具、订单池、校验结果、属性面板以悬浮面板形式覆盖在地图上层
- **DOM 顶部工具条**：导入导出、视图模式切换、重置视角等全局操作使用顶部悬浮工具条承载

这种架构的优势：

1. DOM 适合表单、按钮等 UI 组件，开发效率高
2. Canvas 适合地图、对象和网格渲染，与游戏主体保持一致
3. DOM 与 Canvas 分层明确，便于实现拖动、折叠、停靠、状态恢复等编辑器交互
4. 地图的缩放与浏览由 Phaser 相机统一负责，避免 DOM 滚动和画布缩放相互干扰

### 工具面板

工具区不再固定在左侧，而是拆成多个可移动浮窗。默认布局如下：

- 左侧停靠：`基础设置`、`目标分数`、`工具面板`、`订单池`
- 右侧停靠：`属性面板`、`校验结果`
- 面板支持拖动、最小化、展开、内部滚动
- 面板拖到边缘 24px 范围内时自动吸附停靠
- 布局状态保存在 `level-editor-layout-v1`

### 地图编辑区域

#### 特性

1. **网格系统**
   - 地图世界坐标基于格子系统，支持自定义宽高
   - 每格 48x48 像素
   - 显示坐标辅助线

2. **视图模式**
   - `适配视图`：默认模式，相机自动缩放到完整展示整张地图
   - `浏览视图`：按正常比例查看局部区域，支持缩放和平移
   - `重置视角`：根据当前视图模式恢复默认相机状态

3. **交互操作**
   - 点击工具面板中的工具后，在地图上点击放置对象
   - 点击选中地图上的对象
   - 按 Delete/Backspace 删除选中对象
   - 浏览视图下鼠标滚轮缩放
   - 浏览视图下按住空格拖动画布平移相机
   - 拖动浮窗时地图停止响应放置和选择事件，避免交互冲突

4. **视觉反馈**
   - 使用 Phaser 渲染地板、工作站、玩家和网格
   - 选中对象时显示高亮边框
   - 传送带显示方向箭头
   - 工作站和食材箱显示中文标签

5. **当前实现范围**
   - 已实现：全屏地图、视图切换、滚轮缩放、空格平移、地图尺寸联动、局部交互屏蔽
   - 规划中：撤销/重做、复制/粘贴、试玩模式

```typescript
  // 1. 清空现有游戏对象
  ALL_PLAYERS.length = 0;
  ALL_ITEMS.length = 0;
  ALL_STATIONS.length = 0;
  
  // 2. 设置游戏参数
  scene.cutDownTime = config.duration;
  scene.scoreTarget = config.scoreTarget;
  
  // 3. 初始化物理世界边界
  scene.physics.world.setBounds(
    0,
    0,
    TILE_SIZE * config.map.width,
    TILE_SIZE * config.map.height,
  );
  
  // 4. 创建地板
  createFloorsFromConfig(scene, config.map.floors);
  
  // 5. 创建工作站
  createStationsFromConfig(scene, config.stations);
  
  // 6. 创建玩家
  createPlayersFromConfig(scene, config.players);
  
  // 7. 配置订单池
  orderManager.reset();
  orderManager.setRecipePool(
    config.orderPool.recipes.map(id => getRecipeById(id)).filter(Boolean)
  );
  
  // 8. 设置碰撞检测
  handleCollision(scene);
  handleThrow(scene);
}

 * 创建地板

function createFloorsFromConfig(
  scene: Phaser.Scene, 
  floors: FloorConfig[]
): void {
  floors.forEach(floorConfig => {
    const x = floorConfig.x * TILE_SIZE + TILE_SIZE / 2;
    const y = floorConfig.y * TILE_SIZE + TILE_SIZE / 2;
    
    switch (floorConfig.type) {
      case 'normal':
        createNormalFloor(scene, x, y, floorConfig);
        break;
      case 'wall':
        createWallFloor(scene, x, y, floorConfig);
        break;
      case 'conveyor':
        createConveyorFloor(scene, x, y, floorConfig as ConveyorFloor);
        break;
    }
  });
}

function createNormalFloor(
  scene: Phaser.Scene,
  x: number,
  y: number,
  config: NormalFloor
): void {
  const texture = config.texture || 'floor';
  scene.add.image(x, y, texture).setDepth(DEPTH.FLOOR);
}

function createWallFloor(
  scene: Phaser.Scene,
  x: number,
  y: number,
  config: WallFloor
): void {
  const texture = config.texture || 'wall';
  const wall = scene.physics.add.staticSprite(x, y, texture);
  wall.setDepth(DEPTH.FLOOR);
  // 墙壁需要碰撞体
  wall.body.setSize(TILE_SIZE, TILE_SIZE);
}

function createConveyorFloor(
  scene: Phaser.Scene,
  x: number,
  y: number,
  config: ConveyorFloor
): void {
  const texture = config.texture || 'conveyor';
  const conveyor = scene.add.image(x, y, texture);
  conveyor.setDepth(DEPTH.FLOOR);
  
  // 根据方向旋转纹理
  const rotationMap = {
    'up': -90,
    'down': 90,
    'left': 180,
    'right': 0
  };
  conveyor.setAngle(rotationMap[config.direction] || 0);
  
  // 存储传送带配置到场景数据中，供物理更新使用
  if (!scene.data.has('conveyors')) {
    scene.data.set('conveyors', []);
  }
  const conveyors = scene.data.get('conveyors') as ConveyorFloor[];
  conveyors.push(config);
}
```

### 传送带物理更新

传送带需要在游戏循环中持续更新玩家和物品的位置。

```typescript
/**
 * 在 GameScene.update() 中调用
 */
function updateConveyors(scene: Phaser.Scene, delta: number): void {
  const conveyors = scene.data.get('conveyors') as ConveyorFloor[] | undefined;
  if (!conveyors) return;
  
  conveyors.forEach(conveyor => {
    const conveyorX = conveyor.x * TILE_SIZE + TILE_SIZE / 2;
    const conveyorY = conveyor.y * TILE_SIZE + TILE_SIZE / 2;
    const halfTile = TILE_SIZE / 2;
    
    // 检查玩家是否在传送带上
    ALL_PLAYERS.forEach(player => {
      const dx = Math.abs(player.x - conveyorX);
      const dy = Math.abs(player.y - conveyorY);
      
      if (dx < halfTile && dy < halfTile) {
        // 玩家在传送带上，应用推力
        applyConveyorForce(player, conveyor, delta);
      }
    });
    
    // 检查物品是否在传送带上
    ALL_ITEMS.forEach(item => {
      if (item.isFlying || item.heldBy) return; // 飞行或被持有的物品不受传送带影响
      
      const dx = Math.abs(item.x - conveyorX);
      const dy = Math.abs(item.y - conveyorY);
      
      if (dx < halfTile && dy < halfTile) {
        applyConveyorForce(item, conveyor, delta);
      }
    });
  });
}

function applyConveyorForce(
  object: Phaser.Physics.Arcade.Sprite,
  conveyor: ConveyorFloor,
  delta: number
): void {
  const force = (conveyor.speed * delta) / 1000; // 转换为每帧的移动量
  
  switch (conveyor.direction) {
    case 'up':
      object.y -= force;
      break;
    case 'down':
      object.y += force;
      break;
    case 'left':
      object.x -= force;
      break;
    case 'right':
      object.x += force;
      break;
  }
}
```

```typescript
function exportLevelConfig(scene: Phaser.Scene): LevelConfig {
  // 收集地板
  const floors: FloorConfig[] = [];
  const floorSprites = scene.children.list.filter(
    obj => obj.depth === DEPTH.FLOOR
  );
  
  floorSprites.forEach((sprite: any) => {
    const floorData = sprite.getData('floorConfig');
    if (floorData) {
      floors.push(floorData);
    }
  });
  
  // 收集工作站
  const stations: StationConfig[] = ALL_STATIONS.map(station => {
    const baseConfig: BaseStationConfig = {
      type: station.type,
      x: Math.round((station.x - TILE_SIZE / 2) / TILE_SIZE),
      y: Math.round((station.y - TILE_SIZE / 2) / TILE_SIZE),
    };
    
    if (station.sprite.angle !== 0) {
      baseConfig.rotation = station.sprite.angle;
    }
    
    // 根据工作站类型添加特定属性
    switch (station.type) {
      case 'ingredient':
        return {
          ...baseConfig,
          ingredientType: (station as IngredientStation).ingredientType,
          infinite: (station as IngredientStation).infinite,
          maxCount: (station as IngredientStation).maxCount,
        } as IngredientStation;
      
      case 'pot':
        return {
          ...baseConfig,
          cookSpeed: (station as PotStation).workSpeed,
          canBurn: (station as PotStation).canBurn,
          canFire: (station as PotStation).canFire,
        } as PotStation;
      
      // 其他类型类似处理...
      default:
        return baseConfig as StationConfig;
    }
  });
  
  // 收集玩家
  const players: PlayerSpawn[] = ALL_PLAYERS.map(player => ({
    id: player.id,
    x: Math.round((player.x - TILE_SIZE / 2) / TILE_SIZE),
    y: Math.round((player.y - TILE_SIZE / 2) / TILE_SIZE),
    color: player.tintTopLeft,
  }));
  
  // 构建完整配置
  const config: LevelConfig = {
    id: generateLevelId(),
    name: '未命名关卡',
    version: '1.0',
    gameType: 'local-coop',
    duration: scene.cutDownTime || 300,
    scoreTarget: scene.scoreTarget || {
      star1: 100,
      star2: 200,
      star3: 300,
    },
    map: {
      width: Math.ceil(scene.physics.world.bounds.width / TILE_SIZE),
      height: Math.ceil(scene.physics.world.bounds.height / TILE_SIZE),
      floors,
    },
    players,
    stations,
    orderPool: {
      recipes: orderManager.getAllRecipeIds(),
      maxActiveOrders: 4,
      spawnInterval: 15,
    },
  };
  
  return config;
}
```

```typescript
interface ValidationResult {
  valid: boolean;
  errors: string[];
  warnings: string[];
}

function validateLevel(config: LevelConfig): ValidationResult {
  const errors: string[] = [];
  const warnings: string[] = [];
  
  // 必须有至少一个玩家
  if (config.players.length === 0) {
    errors.push('关卡必须至少有一个玩家出生点');
  }
  
  // 必须有上菜口
  const hasDelivery = config.stations.some(s => s.type === 'delivery');
  if (!hasDelivery) {
    errors.push('关卡必须有至少一个上菜口');
  }
  
  // 必须有盘子来源
  const hasPlate = config.stations.some(s => s.type === 'plate-counter');
  if (!hasPlate) {
    warnings.push('关卡没有盘子柜台，玩家无法获取盘子');
  }
  
  // 检查订单池中的菜谱是否需要关卡中没有的食材
  const availableIngredients = new Set(
    config.stations
      .filter(s => s.type === 'ingredient')
      .map(s => s.ingredientType)
  );
  
  config.orderPool.recipes.forEach(recipeId => {
    const recipe = getRecipeById(recipeId);
    if (recipe) {
      const required = getRequiredIngredients(recipe);
      const missing = required.filter(ing => !availableIngredients.has(ing));
      if (missing.length > 0) {
        warnings.push(
          `菜谱"${recipe.displayName}"需要的食材"${missing.join('、')}"在关卡中不可用`
        );
      }
    }
  });
  
  return {
    valid: errors.length === 0,
    errors,
    warnings
  };
}
```
